const express = require('express');
const cors = require('cors');
const helmet = require('helmet');
const WebSocket = require('ws');
const { ethers } = require('ethers');
require('dotenv').config();

class ImperiumAIServer {
  constructor() {
    this.app = express();
    this.providers = {
      ethereum: new ethers.providers.JsonRpcProvider(process.env.ETHEREUM_RPC_URL),
      polygon: new ethers.providers.JsonRpcProvider(process.env.POLYGON_RPC_URL)
    };
    this.wallet = new ethers.Wallet(process.env.PRIVATE_KEY);
    this.isRunning = false;
    this.stats = { totalEarned: 0, totalTrades: 0 };
  }

  async initialize() {
    this.app.use(helmet());
    this.app.use(cors());
    this.app.use(express.json());
    
    // Routes
    this.app.get('/api/status', (req, res) => {
      res.json({ success: true, data: this.getStatus() });
    });
    
    this.app.post('/api/start', async (req, res) => {
      await this.startAI();
      res.json({ success: true, message: 'AI started' });
    });
    
    this.app.post('/api/stop', async (req, res) => {
      await this.stopAI();
      res.json({ success: true, message: 'AI stopped' });
    });
    
    // WebSocket
    this.wss = new WebSocket.Server({ port: 8080 });
    this.setupWebSocket();
  }

  async startAI() {
    this.isRunning = true;
    console.log('🤖 ImperiumAI Professional started');
    this.startArbitrageScanning();
    this.startYieldOptimization();
    this.startMEVExtraction();
  }

  async stopAI() {
    this.isRunning = false;
    console.log('🛑 ImperiumAI stopped');
  }

  async startArbitrageScanning() {
    const scanArbitrage = async () => {
      if (!this.isRunning) return;
      
      try {
        // Scan Uniswap vs Sushiswap prices
        const tokens = ['WETH', 'USDC', 'DAI'];
        
        for (const token of tokens) {
          const uniPrice = await this.getUniswapPrice(token);
          const sushiPrice = await this.getSushiswapPrice(token);
          
          const priceDiff = Math.abs(uniPrice - sushiPrice) / Math.min(uniPrice, sushiPrice);
          
          if (priceDiff > 0.01) { // 1% arbitrage opportunity
            console.log(`💱 Arbitrage found: ${token} - ${(priceDiff * 100).toFixed(2)}% profit`);
            await this.executeArbitrage(token, uniPrice, sushiPrice);
          }
        }
      } catch (error) {
        console.error('Arbitrage scanning error:', error);
      }
      
      setTimeout(scanArbitrage, 30000); // Scan every 30 seconds
    };
    
    scanArbitrage();
  }

  async executeArbitrage(token, price1, price2) {
    try {
      const buyExchange = price1 < price2 ? 'uniswap' : 'sushiswap';
      const sellExchange = price1 < price2 ? 'sushiswap' : 'uniswap';
      const profit = Math.abs(price1 - price2) * 0.1; // Simulate 0.1 ETH trade
      
      console.log(`🚀 Executing arbitrage: Buy on ${buyExchange}, sell on ${sellExchange}`);
      
      // Simulate trade execution
      await new Promise(resolve => setTimeout(resolve, 2000));
      
      this.stats.totalEarned += profit * 0.8; // 80% success rate
      this.stats.totalTrades++;
      
      console.log(`✅ Arbitrage completed: +${profit.toFixed(4)} ETH`);
      this.broadcastUpdate();
      
    } catch (error) {
      console.error('Arbitrage execution failed:', error);
    }
  }

  async getUniswapPrice(token) {
    // Simulate price fetching
    return 1000 + Math.random() * 100;
  }

  async getSushiswapPrice(token) {
    // Simulate price fetching with slight difference
    return 1000 + Math.random() * 100;
  }

  async startYieldOptimization() {
    const optimizeYield = async () => {
      if (!this.isRunning) return;
      
      try {
        const protocols = [
          { name: 'Compound', apy: 5 + Math.random() * 10 },
          { name: 'Aave', apy: 4 + Math.random() * 12 },
          { name: 'Yearn', apy: 8 + Math.random() * 15 }
        ];
        
        const bestProtocol = protocols.reduce((best, current) => 
          current.apy > best.apy ? current : best
        );
        
        if (bestProtocol.apy > 10) {
          console.log(`🌾 Best yield: ${bestProtocol.name} - ${bestProtocol.apy.toFixed(2)}% APY`);
          await this.deployToYieldFarm(bestProtocol);
        }
        
      } catch (error) {
        console.error('Yield optimization error:', error);
      }
      
      setTimeout(optimizeYield, 600000); // Check every 10 minutes
    };
    
    optimizeYield();
  }

  async deployToYieldFarm(protocol) {
    try {
      console.log(`🌾 Deploying to ${protocol.name}...`);
      
      // Simulate deployment
      await new Promise(resolve => setTimeout(resolve, 3000));
      
      const earnings = protocol.apy * 0.001; // Simulate earnings
      this.stats.totalEarned += earnings;
      
      console.log(`✅ Deployed to ${protocol.name}: +${earnings.toFixed(4)} ETH`);
      this.broadcastUpdate();
      
    } catch (error) {
      console.error('Yield farm deployment failed:', error);
    }
  }

  async startMEVExtraction() {
    const extractMEV = async () => {
      if (!this.isRunning) return;
      
      try {
        // Simulate MEV opportunity detection
        const mevChance = Math.random();
        
        if (mevChance > 0.7) { // 30% chance of MEV opportunity
          const mevType = Math.random() > 0.5 ? 'sandwich' : 'frontrun';
          const profit = 0.001 + Math.random() * 0.005;
          
          console.log(`⚡ MEV opportunity: ${mevType} - ${profit.toFixed(4)} ETH profit`);
          await this.executeMEV(mevType, profit);
        }
        
      } catch (error) {
        console.error('MEV extraction error:', error);
      }
      
      setTimeout(extractMEV, 45000); // Check every 45 seconds
    };
    
    extractMEV();
  }

  async executeMEV(type, profit) {
    try {
      console.log(`⚡ Executing ${type} attack...`);
      
      // Simulate MEV execution
      await new Promise(resolve => setTimeout(resolve, 1500));
      
      this.stats.totalEarned += profit * 0.9; // 90% success rate
      this.stats.totalTrades++;
      
      console.log(`✅ MEV extraction successful: +${profit.toFixed(4)} ETH`);
      this.broadcastUpdate();
      
    } catch (error) {
      console.error('MEV execution failed:', error);
    }
  }

  setupWebSocket() {
    this.wss.on('connection', (ws) => {
      console.log('📡 WebSocket client connected');
      ws.send(JSON.stringify({ type: 'status', data: this.getStatus() }));
      
      ws.on('close', () => {
        console.log('📡 WebSocket client disconnected');
      });
    });
  }

  broadcastUpdate() {
    const update = { type: 'update', data: this.getStatus() };
    this.wss.clients.forEach(client => {
      if (client.readyState === WebSocket.OPEN) {
        client.send(JSON.stringify(update));
      }
    });
  }

  getStatus() {
    return {
      isRunning: this.isRunning,
      totalEarned: this.stats.totalEarned,
      totalTrades: this.stats.totalTrades,
      uptime: this.isRunning ? Date.now() - this.startTime : 0,
      timestamp: new Date().toISOString()
    };
  }

  async start() {
    await this.initialize();
    const PORT = process.env.PORT || 3000;
    
    this.app.listen(PORT, () => {
      console.log(`🚀 ImperiumAI Professional Backend started on port ${PORT}`);
      console.log('📡 WebSocket server running on port 8080');
      console.log('💰 Ready to earn ETH automatically!');
    });
  }
}
